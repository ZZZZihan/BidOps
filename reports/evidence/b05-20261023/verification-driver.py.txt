import datetime
import json
import os
from pathlib import Path
import subprocess
import sys

repo, output = map(lambda p: Path(p).resolve(), sys.argv[1:3])
output.mkdir(parents=True, exist_ok=False)
git = lambda *args: subprocess.check_output(["git", *args], cwd=repo, text=True).strip()
if git("status", "--porcelain"):
    raise SystemExit("verification requires a clean committed worktree")
result = {"code_sha": git("rev-parse", "HEAD"), "tree_sha": git("rev-parse", "HEAD^{tree}"),
          "executor": "root", "source_kind": "synthetic_engineering_and_protocol_fixtures",
          "actual_model_id": None, "model_calls": 0, "estimated_cost_cny": "0", "commands": []}
checks = [
    ("backend", [".venv/bin/pytest", "-q", "--basetemp", str(output / "pytest")], repo),
    ("lint", [".venv/bin/ruff", "check", "."], repo),
    ("format", [".venv/bin/ruff", "format", "--check", "."], repo),
    ("freeze", [".venv/bin/python", "scripts/screening_eval.py",
       "--expected-freeze-sha256", "22a2adf96f07b9a3e9c1493dae2cebbe2499b6e05db21f2b52855686c2afffea",
       "--expected-input-sha256", "eda43a451ae978251de4be003f1c062d1a4d3e93db9249c9c5458966628f44fb",
       "--output", str(output / "frozen-scorer.json")], repo),
    ("frontend-build", ["npm", "run", "build"], repo / "frontend"),
    ("frontend-browser", ["npm", "run", "test:e2e"], repo / "frontend"),
]
env = dict(os.environ, BIDOPS_NETWORK_ENABLED="false", BIDOPS_BROWSER_CHANNEL="chrome")
for name, cmd, cwd in checks:
    before = datetime.datetime.now(datetime.UTC)
    with (output / f"{name}.txt").open("w") as log:
        proc = subprocess.run(cmd, cwd=cwd, env=env, stdout=log, stderr=subprocess.STDOUT)
    after = datetime.datetime.now(datetime.UTC)
    row = {"name": name, "command": cmd, "cwd": str(cwd), "start": before.isoformat(),
           "end": after.isoformat(), "exit_code": proc.returncode, "log": f"{name}.txt"}
    result["commands"].append(row)
    (output / "verification.json").write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n")
    print(json.dumps({"check": name, "exit_code": proc.returncode}, ensure_ascii=False), flush=True)
    if proc.returncode:
        raise SystemExit(proc.returncode)
result["working_tree_clean_after"] = not git("status", "--porcelain")
result["status"] = "PASS" if result["working_tree_clean_after"] else "FAIL"
(output / "verification.json").write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n")
print(json.dumps(result), flush=True)
